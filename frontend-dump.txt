import { defineConfig } from 'vite'
import react from '@vitejs/plugin-react'
import tailwindcss from '@tailwindcss/vite'
import fs from 'node:fs'
import path from 'node:path'

// Face models ship inside the @vladmandic/face-api package (node_modules/@vladmandic/face-api/model).
// This plugin serves them at /models in dev and copies them into dist/models on build,
// so you do not have to download or copy anything by hand.
function faceModels() {
  const dir = path.resolve('node_modules/@vladmandic/face-api/model')
  const types = { '.json': 'application/json', '.bin': 'application/octet-stream' }
  return {
    name: 'face-models',
    configureServer(server) {
      server.middlewares.use('/models', (req, res, next) => {
        const file = path.join(dir, decodeURIComponent((req.url || '').split('?')[0]))
        if (!file.startsWith(dir) || !fs.existsSync(file) || !fs.statSync(file).isFile()) return next()
        res.setHeader('Content-Type', types[path.extname(file)] || 'application/octet-stream')
        fs.createReadStream(file).pipe(res)
      })
    },
    closeBundle() {
      if (!fs.existsSync(dir)) return
      const out = path.resolve('dist/models')
      fs.mkdirSync(out, { recursive: true })
      for (const f of fs.readdirSync(dir)) {
        if (/\.(json|bin)$/.test(f)) fs.copyFileSync(path.join(dir, f), path.join(out, f))
      }
    },
  }
}

// In dev, /api is proxied to the backend, so no CORS setup is needed locally.
export default defineConfig({
  plugins: [react(), tailwindcss(), faceModels()],
  server: { port: 5173, proxy: { '/api': 'http://localhost:5000' } },
})


===== src\api.js =====
// All backend calls go through api(). Backend shape: { success, data, message, errors: [{ field, message }] }
const BASE = import.meta.env.VITE_API_URL || (import.meta.env.PROD ? 'https://hcm-lite-backend.vercel.app/api/v1' : '/api/v1')
let token = localStorage.getItem('token')

export const getToken = () => token
export const setToken = (t) => {
  token = t
  if (t) localStorage.setItem('token', t)
  else localStorage.removeItem('token')
}

export class ApiError extends Error {
  constructor(status, message, errors = []) {
    super(message)
    this.status = status
    this.errors = errors
  }
}

export async function api(path, { method = 'GET', body } = {}) {
  let res
  try {
    res = await fetch(BASE + path, {
      method,
      headers: { 'Content-Type': 'application/json', ...(token ? { Authorization: `Bearer ${token}` } : {}) },
      body: body ? JSON.stringify(body) : undefined,
    })
  } catch {
    throw new ApiError(0, 'Cannot reach the server. Check that the backend is running.')
  }
  const json = await res.json().catch(() => ({}))
  if (!res.ok) {
    if (res.status === 401 && token && !path.startsWith('/auth/login')) {
      setToken(null)
      window.dispatchEvent(new Event('auth:logout'))
    }
    throw new ApiError(res.status, json.message || 'Request failed', json.errors || [])
  }
  return json.data
}

// Downloads a file (for example a payslip PDF). A plain link cannot send the login token, so we fetch it.
export async function download(path, filename) {
  let res
  try {
    res = await fetch(BASE + path, { headers: token ? { Authorization: `Bearer ${token}` } : {} })
  } catch {
    throw new ApiError(0, 'Cannot reach the server. Check that the backend is running.')
  }
  if (!res.ok) {
    const json = await res.json().catch(() => ({}))
    throw new ApiError(res.status, json.message || 'Download failed')
  }
  const url = URL.createObjectURL(await res.blob())
  const a = document.createElement('a')
  a.href = url
  a.download = filename
  document.body.appendChild(a)
  a.click()
  a.remove()
  URL.revokeObjectURL(url)
}

===== src\auth.jsx =====
import { createContext, useCallback, useContext, useEffect, useState } from 'react'
import { api, getToken, setToken } from './api.js'

const Ctx = createContext(null)
export const useAuth = () => useContext(Ctx)

export function AuthProvider({ children }) {
  const [user, setUser] = useState(() => {
    try { return getToken() ? JSON.parse(localStorage.getItem('user')) : null } catch { return null }
  })

  const apply = useCallback(({ token, user }) => {
    setToken(token)
    localStorage.setItem('user', JSON.stringify(user))
    setUser(user)
  }, [])

  const logout = useCallback(() => {
    setToken(null)
    localStorage.removeItem('user')
    setUser(null)
  }, [])

  useEffect(() => {
    window.addEventListener('auth:logout', logout)
    return () => window.removeEventListener('auth:logout', logout)
  }, [logout])

  const login = async (email, password) => {
    const data = await api('/auth/login', { method: 'POST', body: { email, password } })
    apply(data)
    return data.user
  }

  return <Ctx.Provider value={{ user, login, logout, apply }}>{children}</Ctx.Provider>
}

import { useEffect, useRef, useState } from 'react'
import { api } from '../api.js'
import { Btn, inputCls } from './bits.jsx'

const HINTS = {
  employee: ['How many sick leaves do I have left this year?', 'Can I take leave on a public holiday?'],
  hr: ['Who is on leave today?', 'Who came late most often this month?', 'What is this month’s attendance rate?'],
  admin: ['How many active employees do we have?', 'What is the payroll summary for this month?'],
}

// Floating assistant. Opens from the button or from anywhere via window.dispatchEvent(new Event('ai:open')).
export default function AiChat({ role }) {
  const [open, setOpen] = useState(false)
  const [msgs, setMsgs] = useState([])
  const [text, setText] = useState('')
  const [busy, setBusy] = useState(false)
  const end = useRef(null)

  useEffect(() => {
    const on = () => setOpen(true)
    window.addEventListener('ai:open', on)
    return () => window.removeEventListener('ai:open', on)
  }, [])
  useEffect(() => { end.current?.scrollIntoView({ behavior: 'smooth' }) }, [msgs, busy])

  const send = async (q) => {
    const message = (q ?? text).trim()
    if (!message || busy) return
    setText('')
    const history = msgs.filter((m) => !m.error).map((m) => ({ role: m.role, content: m.content }))
    setMsgs((m) => [...m, { role: 'user', content: message }])
    setBusy(true)
    try {
      const res = await api('/ai/chat', { method: 'POST', body: { message, history } })
      setMsgs((m) => [...m, { role: 'assistant', content: res.answer }])
    } catch (e) {
      setMsgs((m) => [...m, { role: 'assistant', content: e.message, error: true }])
    } finally { setBusy(false) }
  }

  if (!open) {
    return <button onClick={() => setOpen(true)} className="fixed bottom-5 right-5 z-40 rounded-full bg-pine-dark px-4 py-3 text-sm font-medium text-white shadow-lg hover:opacity-90">Ask assistant</button>
  }
  return (
    <div className="fixed inset-x-3 bottom-3 z-40 flex max-h-[80vh] flex-col rounded-lg border border-sand bg-white shadow-xl sm:inset-x-auto sm:right-5 sm:bottom-5 sm:w-96">
      <div className="flex items-center justify-between border-b border-sand px-4 py-2">
        <div><div className="font-display font-bold">Assistant</div><div className="text-[11px] text-gray-400">AI-generated. Answers come from your company data.</div></div>
        <button onClick={() => setOpen(false)} aria-label="Close" className="rounded px-2 text-xl leading-none hover:bg-sand/60">×</button>
      </div>
      <div className="min-h-[200px] flex-1 space-y-3 overflow-y-auto p-4 text-sm">
        {msgs.length === 0 && (
          <div>
            <p className="text-gray-500">Ask about your data or company policy. You can write in English or Roman Urdu.</p>
            <div className="mt-2 flex flex-col gap-1.5">
              {(HINTS[role] || []).map((h) => <button key={h} onClick={() => send(h)} className="rounded-md border border-sand px-2 py-1.5 text-left hover:bg-sand/50">{h}</button>)}
            </div>
          </div>
        )}
        {msgs.map((m, i) => (
          <div key={i} className={m.role === 'user' ? 'text-right' : ''}>
            <div className={`inline-block max-w-[90%] whitespace-pre-wrap rounded-lg px-3 py-2 text-left ${m.role === 'user' ? 'bg-pine-dark text-white' : m.error ? 'bg-red-50 text-red-800' : 'bg-sand/60'}`}>{m.content}</div>
            {m.role === 'assistant' && !m.error && <div className="mt-0.5 text-[10px] text-gray-400">AI-generated</div>}
          </div>
        ))}
        {busy && <div className="text-xs text-gray-400">Thinking...</div>}
        <div ref={end} />
      </div>
      <form className="flex gap-2 border-t border-sand p-3" onSubmit={(e) => { e.preventDefault(); send() }}>
        <input className={inputCls} placeholder="Type your question" value={text} maxLength={1000} onChange={(e) => setText(e.target.value)} />
        <Btn disabled={busy || !text.trim()}>Send</Btn>
      </form>
    </div>
  )
}

// Loads data on mount and whenever deps change. reload() refreshes.
export function useLoad(fn, deps = []) {
  const [data, setData] = useState(null)
  const [error, setError] = useState(null)
  const [loading, setLoading] = useState(true)
  const run = useCallback(async () => {
    setLoading(true)
    try { setData(await fn()); setError(null) } catch (e) { setError(e) } finally { setLoading(false) }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, deps)
  useEffect(() => { run() }, [run])
  return { data, error, loading, reload: run, setError }
}

// Counts a number up once when it appears (skipped if the user prefers reduced motion).
function useCountUp(target, duration = 700) {
  const [n, setN] = useState(typeof target === 'number' ? 0 : target)
  useEffect(() => {
    if (typeof target !== 'number' || window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
      setN(target)
      return undefined
    }
    let raf
    const t0 = performance.now()
    const tick = (now) => {
      const p = Math.min(1, (now - t0) / duration)
      setN(target * (1 - Math.pow(1 - p, 3)))
      if (p < 1) raf = requestAnimationFrame(tick)
    }
    raf = requestAnimationFrame(tick)
    return () => cancelAnimationFrame(raf)
  }, [target, duration])
  return n
}

// Handles 12, "12%", "Rs 1,200". Anything else ("Unlimited", "-") is shown as is.
const PARSE = /^(\D*?)(\d[\d,]*(?:\.\d+)?)(\D*)$/

function AnimatedValue({ value }) {
  const m = typeof value === 'string' ? PARSE.exec(value) : null
  const target = typeof value === 'number' ? value : m ? Number(m[2].replace(/,/g, '')) : null
  const decimals = m ? (m[2].split('.')[1] || '').length : typeof value === 'number' && !Number.isInteger(value) ? 2 : 0
  const n = useCountUp(target)
  if (target == null) return <>{value ?? '-'}</>
  const text = Number(n).toLocaleString('en-PK', { minimumFractionDigits: decimals, maximumFractionDigits: decimals })
  return <>{m ? m[1] : ''}{text}{m ? m[3] : ''}</>
}

// Usage:
//   const [confirm, confirmDialog] = useConfirm()
//   if (!(await confirm({ title: 'Delete holiday?', message: '...', confirmLabel: 'Delete' }))) return
//   ...and render {confirmDialog} once in the component's JSX.
export function useConfirm() {
  const [state, setState] = useState(null)
  const resolver = useRef(null)

  const ask = useCallback(
    (opts) =>
      new Promise((resolve) => {
        resolver.current = resolve
        setState(opts)
      }),
    []
  )
  const close = (answer) => {
    resolver.current?.(answer)
    resolver.current = null
    setState(null)
  }

  const dialog = state && (
    <Modal title={state.title || 'Are you sure?'} onClose={() => close(false)} size="sm">
      <p className="text-sm text-ink/80">{state.message}</p>
      <div className="mt-5 flex justify-end gap-2">
        <Btn variant="ghost" autoFocus onClick={() => close(false)}>Cancel</Btn>
        <Btn variant="dangerSolid" onClick={() => close(true)}>{state.confirmLabel || 'Delete'}</Btn>
      </div>
    </Modal>
  )
  return [ask, dialog]
}

// Opens the webcam, finds one face and returns its 128-number descriptor.
// mode "enroll" takes 3 samples and averages them; "verify" takes 1.
// Model files are served at /models (see vite.config.js).
// Tries the local /models folder first, then falls back to public CDNs, so it works
// even if the model files were never copied into the project.
const MODEL_SOURCES = [
  '/models',
  'https://cdn.jsdelivr.net/npm/@vladmandic/face-api@1.7.15/model',
  'https://vladmandic.github.io/face-api/model',
]

// A source is usable only if it returns real JSON (not the app's index.html).
// Returns null when fine, otherwise a short reason.
const checkSource = async (base) => {
  try {
    const r = await fetch(`${base}/tiny_face_detector_model-weights_manifest.json`)
    if (!r.ok) return `file not found (HTTP ${r.status})`
    JSON.parse(await r.text())
    return null
  } catch (e) { return `not reachable or not JSON (${e?.message || e})` }
}

let apiPromise
const loadFaceApi = () => {
  apiPromise ||= import('@vladmandic/face-api')
    .then(async (faceapi) => {
      const problems = []
      for (const base of MODEL_SOURCES) {
        const bad = await checkSource(base)
        if (bad) { problems.push(`${base}: ${bad}`); continue }
        try {
          await faceapi.tf?.ready?.()
          await Promise.all([
            faceapi.nets.tinyFaceDetector.loadFromUri(base),
            faceapi.nets.faceLandmark68Net.loadFromUri(base),
            faceapi.nets.faceRecognitionNet.loadFromUri(base),
          ])
          return faceapi
        } catch (e) { problems.push(`${base}: loading failed (${e?.message || e})`) }
      }
      throw new Error(`Face models could not be loaded. ${problems.join(' | ')}`)
    })
    .catch((e) => { apiPromise = undefined; throw e }) // allow a retry after a failure
  return apiPromise
}

export default function FaceCapture({ mode = 'verify', onDone, onCancel }) {
  const video = useRef(null)
  const stream = useRef(null)
  const [msg, setMsg] = useState('Starting camera...')
  const [ready, setReady] = useState(false)
  const [busy, setBusy] = useState(false)

  useEffect(() => {
    let dead = false
    ;(async () => {
      try {
        const [faceapi, media] = await Promise.all([loadFaceApi(), navigator.mediaDevices.getUserMedia({ video: { facingMode: 'user', width: 480 } })])
        if (dead) return media.getTracks().forEach((t) => t.stop())
        stream.current = media
        video.current.srcObject = media
        await video.current.play()
        setReady(true)
        setMsg('Look at the camera, then press Scan.')
        video.current.faceapi = faceapi
      } catch (e) {
        setMsg(e?.name === 'NotAllowedError' ? 'Camera permission denied. Allow camera access and try again.' : `Could not start the camera or face models: ${e?.message || e}`)
      }
    })()
    return () => { dead = true; stream.current?.getTracks().forEach((t) => t.stop()) }
  }, [])

  const sample = async () => {
    const faceapi = video.current.faceapi
    const found = await faceapi.detectAllFaces(video.current, new faceapi.TinyFaceDetectorOptions({ inputSize: 320, scoreThreshold: 0.5 })).withFaceLandmarks().withFaceDescriptors()
    if (found.length === 0) throw new Error('No face found. Move closer and face the camera.')
    if (found.length > 1) throw new Error('More than one face in view. Only one person please.')
    return Array.from(found[0].descriptor)
  }

  const scan = async () => {
    setBusy(true)
    try {
      const n = mode === 'enroll' ? 3 : 1
      const all = []
      for (let i = 0; i < n; i++) {
        setMsg(n > 1 ? `Scanning ${i + 1} of ${n}...` : 'Scanning...')
        all.push(await sample())
        if (i < n - 1) await new Promise((r) => setTimeout(r, 500))
      }
      const avg = all[0].map((_, i) => all.reduce((s, d) => s + d[i], 0) / all.length)
      stream.current?.getTracks().forEach((t) => t.stop())
      onDone(avg)
    } catch (e) {
      setMsg(e.message)
      setBusy(false)
    }
  }

  return (
    <div className="mb-5 rounded-lg border border-sand bg-white p-4">
      <div className="mb-2 text-sm font-medium">{mode === 'enroll' ? 'Register face' : 'Verify face'}</div>
      <video ref={video} muted playsInline className="mb-3 aspect-[4/3] w-full max-w-sm rounded-md bg-black object-cover" style={{ transform: 'scaleX(-1)' }} />
      <div className="mb-3 text-sm text-gray-600">{msg}</div>
      <div className="flex gap-2">
        <Btn disabled={!ready || busy} onClick={scan}>Scan</Btn>
        <Btn variant="ghost" onClick={onCancel}>Cancel</Btn>
      </div>
    </div>
  )
}

// Usage: const toast = useToast(); toast.success('Saved'); toast.error(err.message)
const Ctx = createContext({ success: () => {}, error: () => {} })
export const useToast = () => useContext(Ctx)

export function ToastProvider({ children }) {
  const [items, setItems] = useState([])
  const nextId = useRef(0)

  const dismiss = useCallback((id) => setItems((list) => list.filter((t) => t.id !== id)), [])
  const push = useCallback(
    (kind, message) => {
      const id = ++nextId.current
      setItems((list) => [...list.slice(-3), { id, kind, message }])
      setTimeout(() => dismiss(id), kind === 'error' ? 5500 : 3200)
    },
    [dismiss]
  )
  const toast = useMemo(() => ({ success: (m) => push('success', m), error: (m) => push('error', m) }), [push])

  return (
    <Ctx.Provider value={toast}>
      {children}
      <div className="pointer-events-none fixed right-4 top-4 z-[60] flex w-[min(92vw,22rem)] flex-col gap-2">
        {items.map((t) => (
          <div
            key={t.id}
            role={t.kind === 'error' ? 'alert' : 'status'}
            className={cx(
              'anim-toast pointer-events-auto flex items-start gap-3 rounded-2xl border bg-white px-4 py-3 text-sm shadow-lift',

                'mt-0.5 grid h-5 w-5 shrink-0 place-items-center rounded-full text-xs font-bold text-white',

export function Alert({ kind = 'error', children }) {
  if (!children) return null
  const s = kind === 'error' ? 'border-red-200 bg-red-50 text-red-800' : 'border-amber/30 bg-amber/10 text-amber'
  return <div role="alert" className={cx('rounded-md border px-3 py-2 text-sm', s)}>{children}</div>
}

// The modal is rendered into document.body (portal), so parent layout such as "space-y-5"
// can no longer push it down the page. It is always centred in the screen, the page behind
// it does not scroll, and only the modal body scrolls when the content is taller than the screen.
export function Modal({ title, onClose, footer, children }) {
  useEffect(() => {
    const prev = document.body.style.overflow
    document.body.style.overflow = 'hidden'
    const onKey = (e) => { if (e.key === 'Escape') onClose() }
    window.addEventListener('keydown', onKey)
    return () => {
      document.body.style.overflow = prev
      window.removeEventListener('keydown', onKey)
    }
  }, [onClose])

  return createPortal(
    <div className="fixed inset-0 z-50 flex items-center justify-center bg-ink/40 p-3 sm:p-6">
      <div role="dialog" aria-modal="true" className="flex max-h-[90vh] w-full max-w-2xl flex-col rounded-lg bg-white shadow-xl">
        <div className="flex shrink-0 items-center justify-between border-b border-sand px-5 py-3">
          <h2 className="text-lg font-semibold">{title}</h2>
          <button onClick={onClose} aria-label="Close" className="rounded px-2 text-xl leading-none hover:bg-sand/60">&times;</button>
        </div>
        <div className="overflow-y-auto p-5">{children}</div>
        {footer && <div className="flex shrink-0 justify-end gap-2 border-t border-sand bg-white px-5 py-3">{footer}</div>}
      </div>
    </div>,
    document.body
  )
}

// Some endpoints return `id`, some `_id`. Accept both so delete never calls /announcements/undefined.

function TrashIcon() {
  return (
    <svg viewBox="0 0 24 24" width="16" height="16" fill="none" stroke="currentColor" strokeWidth="1.8" strokeLinecap="round" strokeLinejoin="round" aria-hidden="true">

    </svg>

export default function Announcements() {
  const { user } = useAuth()
  const toast = useToast()
  const [confirm, confirmDialog] = useConfirm()
  const canPost = user.role === 'admin' || user.role === 'hr'
  const canDelete = user.role === 'admin' // add `|| user.role === 'hr'` if the backend allows HR to delete too
  const { data, error, loading, reload } = useLoad(() => api('/announcements'), [])
  const [f, setF] = useState({ title: '', body: '' })
  const [err, setErr] = useState(null)
  const [busy, setBusy] = useState(false)
  const [leaving, setLeaving] = useState(null)

  const post = async (e) => {
    e.preventDefault()
    setBusy(true); setErr(null)
    try {
      await api('/announcements', { method: 'POST', body: f })
      setF({ title: '', body: '' })
      await reload()
      toast.success('Announcement posted')
    } catch (x) { setErr(x) } finally { setBusy(false) }
  }

  const remove = async (a) => {
    const id = idOf(a)
    if (!id) { toast.error('This announcement has no id, so it cannot be deleted.'); return }
    const ok = await confirm({
      title: 'Delete announcement?',
      message: `"${a.title}" will be removed for everyone. This cannot be undone.`,
      confirmLabel: 'Delete',
    })
    if (!ok) return
    try {
      await api(`/announcements/${id}`, { method: 'DELETE' })
      setLeaving(id)
      await new Promise((r) => setTimeout(r, 220)) // let the exit animation play
      await reload()
      toast.success('Announcement deleted')
    } catch (x) {
      toast.error(x.message)
    } finally {
      setLeaving(null)
    }
  }

  return (
    <div>
      <h1 className="mb-4 font-display text-2xl font-bold">Announcements</h1>
      {canPost && (
        <Card title="Post an announcement">
          <form onSubmit={post} className="space-y-3">
            <Field label="Title"><input required minLength="3" className={inputCls} value={f.title} onChange={(e) => setF({ ...f, title: e.target.value })} /></Field>
            <Field label="Message"><textarea required minLength="3" rows="3" className={inputCls} value={f.body} onChange={(e) => setF({ ...f, body: e.target.value })} /></Field>
            <Err error={err} />
            <Btn disabled={busy}>{busy ? 'Posting...' : 'Post'}</Btn>
          </form>
        </Card>
      )}

      <Card title="All announcements">
        <Err error={error} />
        {loading && !data && (
          <div className="space-y-3"><Skeleton className="h-20" /><Skeleton className="h-20" /></div>
        )}
        {data?.length === 0 && (
          <p className="py-6 text-center text-sm text-ink/60">
            {canPost ? 'No announcements yet. Post the first one above.' : 'Nothing has been posted yet.'}
          </p>
        )}
        <ul className="space-y-3">
          {(data || []).map((a) => {
            const id = idOf(a)
            return (
              <li key={id} className={cx('flex gap-3 rounded-xl border border-sand p-4 text-sm', leaving === id && 'anim-leave')}>
                <span className="grid h-9 w-9 shrink-0 place-items-center rounded-full bg-pine-dark text-sm font-semibold text-white" aria-hidden="true">{initial(a.authorName)}</span>
                <div className="min-w-0 flex-1">
                  <div className="font-medium">{a.title}</div>
                  <div className="mt-0.5 whitespace-pre-wrap text-ink/80">{a.body}</div>
                  <div className="mt-1.5 text-xs text-ink/50">{a.authorName} ({a.authorRole}) &middot; {new Date(a.createdAt).toLocaleString()}</div>
                </div>
                {canDelete && (
                  <button
                    onClick={() => remove(a)}
                    aria-label={`Delete announcement ${a.title}`}
                    title="Delete"
                    className="h-9 w-9 shrink-0 place-items-center self-start rounded-full text-ink/40 transition-colors hover:bg-red-50 hover:text-red-600"
                    style={{ display: 'grid' }}
                  >
                    <TrashIcon />
                  </button>
                )}
              </li>
            )
          })}
        </ul>
      </Card>
      {confirmDialog}
    </div>
  )
}

function getPosition() {
  return new Promise((resolve) => {
    if (!navigator.geolocation) return resolve({})
    navigator.geolocation.getCurrentPosition(
      (p) => resolve({ lat: p.coords.latitude, lng: p.coords.longitude }),
      () => resolve({}),
      { timeout: 8000, enableHighAccuracy: true }
    )
  })
}

function Calendar({ days }) {
  const offset = new Date(`${days[0].date}T00:00:00Z`).getUTCDay()
  const names = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat']
  const color = { present: 'bg-green-100', late: 'bg-amber-100', half_day: 'bg-orange-100', absent: 'bg-red-100', leave: 'bg-blue-100', holiday: 'bg-purple-100', weekend: 'bg-gray-100' }
  return (
    <div className="grid grid-cols-7 gap-1 text-center text-xs">
      {names.map((n) => <div key={n} className="py-1 font-medium text-gray-500">{n}</div>)}
      {Array.from({ length: offset }).map((_, i) => <div key={`o${i}`} />)}
      {days.map((d) => (
        <div key={d.date} title={d.holiday || d.status || ''} className={`rounded p-1.5 ${color[d.status] || 'bg-white border border-sand'}`}>
          <div className="font-medium">{Number(d.date.slice(8))}</div>
          <div className="truncate text-[10px] text-gray-600">{d.status === 'upcoming' || !d.status ? '' : d.status.replace('_', ' ')}</div>
        </div>
      ))}
    </div>
  )
}

export default function Attendance() {
  const [month, setMonth] = useState(currentMonth())

  const cal = useLoad(() => api(`/attendance/me?month=${month}`), [month])
  const corr = useLoad(() => api('/attendance/corrections'), [])
  const face = useLoad(() => api('/attendance/face/status'), [])
  const [camera, setCamera] = useState(null) // 'checkin' | 'checkout' | 'enroll'
  const [busy, setBusy] = useState(false)
  const [err, setErr] = useState(null)
  const [form, setForm] = useState({ date: '', checkIn: '09:00', checkOut: '18:00', reason: '' })

  const act = async (fn) => {
    setBusy(true); setErr(null)
    try { await fn(); await Promise.all([today.reload(), cal.reload(), corr.reload()]) } catch (e) { setErr(e) } finally { setBusy(false) }
  }
  const doCheckIn = (descriptor) => act(async () => api('/attendance/check-in', { method: 'POST', body: { ...(await getPosition()), ...(descriptor ? { descriptor } : {}) } }))
  const doCheckOut = (descriptor) => act(() => api('/attendance/check-out', { method: 'POST', body: descriptor ? { descriptor } : {} }))
  const onFace = (descriptor) => {

    if (mode === 'checkin') doCheckIn(descriptor)
    else if (mode === 'checkout') doCheckOut(descriptor)
    else act(async () => { await api('/attendance/face/enroll', { method: 'POST', body: { descriptor } }); await face.reload() })
  }
  const enrolled = face.data?.enrolled
  const submit = (e) => {
    e.preventDefault()
    act(async () => {
      const body = { date: form.date, checkIn: form.checkIn, reason: form.reason }
      if (form.checkOut) body.checkOut = form.checkOut
      await api('/attendance/corrections', { method: 'POST', body })

    })
  }

  const rec = today.data?.record
  return (

          </div>
        </div>
      </Card>

      {camera && <FaceCapture mode={camera === 'enroll' ? 'enroll' : 'verify'} onDone={onFace} onCancel={() => setCamera(null)} />}
      {face.data && !enrolled && !camera && (
        <Card title="Face attendance">
          <p className="mb-3 text-sm text-gray-600">Register your face once. After that, check-in and check-out need a face scan.</p>
          <Btn disabled={busy} onClick={() => setCamera('enroll')}>Register my face</Btn>
        </Card>
      )}

            </div>
            <Calendar days={cal.data.days} />
          </>

          <Field label="Date"><input required type="date" className={inputCls} value={form.date} onChange={(e) => setForm({ ...form, date: e.target.value })} /></Field>

          <Field label="Reason"><input required className={inputCls} value={form.reason} onChange={(e) => setForm({ ...form, reason: e.target.value })} /></Field>
          <div className="sm:col-span-4"><Btn disabled={busy}>Send request</Btn></div>
        </form>
      </Card>

      <Card title="My correction requests">
        <Err error={corr.error} />
        <table className="w-full text-sm">
          <thead><tr className="text-left text-xs text-gray-500"><th>Date</th><th>In</th><th>Out</th><th>Reason</th><th>Status</th></tr></thead>

import { useAuth } from '../auth.jsx'
import AuthShell from '../components/AuthShell.jsx'
import { Alert, Btn, Field, fieldErrors } from '../components/ui.jsx'

export default function ChangePassword() {
  const { user, apply } = useAuth()
  const [f, setF] = useState({ currentPassword: '', newPassword: '', confirm: '' })
  const [err, setErr] = useState(null)
  const [busy, setBusy] = useState(false)
  const navigate = useNavigate()
  const set = (k) => (e) => setF({ ...f, [k]: e.target.value })

  const submit = async (e) => {
    e.preventDefault()
    if (f.newPassword !== f.confirm) return setErr({ message: 'The new passwords do not match.', errors: [] })
    setBusy(true); setErr(null)
    try {
      const data = await api('/auth/change-password', {
        method: 'POST',
        body: { currentPassword: f.currentPassword, newPassword: f.newPassword },
      })
      apply(data)
      navigate('/', { replace: true })
    } catch (x) { setErr(x) } finally { setBusy(false) }
  }
  const fe = fieldErrors(err)

  return (
    <AuthShell
      title="Change password"
      subtitle={user.mustChangePassword ? 'You are using a temporary password. Choose a new one to continue.' : 'Choose a new password.'}
    >
      <form onSubmit={submit} className="space-y-4">
        <Field label="Current password" type="password" value={f.currentPassword} onChange={set('currentPassword')} error={fe.currentPassword} required />
        <Field label="New password (8+ characters)" type="password" value={f.newPassword} onChange={set('newPassword')} error={fe.newPassword} required />
        <Field label="Confirm new password" type="password" value={f.confirm} onChange={set('confirm')} required />
        <Alert>{err && !Object.keys(fe).length ? err.message : null}</Alert>
        <div className="flex gap-2">
          <Btn className="flex-1" disabled={busy}>{busy ? 'Saving…' : 'Save password'}</Btn>
          {!user.mustChangePassword && <Btn type="button" variant="ghost" onClick={() => navigate(-1)}>Cancel</Btn>}
        </div>
      </form>
    </AuthShell>
  )
}

// Runs an API action, refreshes the list, and reports the result as a toast.
function useRun(reload, setErr) {
  const toast = useToast()
  return async (fn, successMessage) => {
    setErr(null)
    try {
      await fn()
      await reload()
      if (successMessage) toast.success(successMessage)
    } catch (e) {
      setErr(e)
      toast.error(e.message)
    }
  }
}

import { useCallback, useEffect, useState } from 'react'
import { api } from '../api.js'
import { Alert, Btn, Card, Empty, Field, fieldErrors } from '../components/ui.jsx'

// Departments and designations are both a simple list of names.
export default function NamedList({ title, path, noun }) {
  const [items, setItems] = useState(null)
  const [name, setName] = useState('')
  const [err, setErr] = useState(null)
  const [busy, setBusy] = useState(false)

  const load = useCallback(() => api(path).then(setItems).catch(setErr), [path])
  useEffect(() => { load() }, [load])

  const add = async (e) => {
    e.preventDefault()
    setBusy(true); setErr(null)
    try { await api(path, { method: 'POST', body: { name } }); setName(''); await load() }
    catch (x) { setErr(x) } finally { setBusy(false) }
  }
  const remove = async (item) => {
    if (!confirm(`Delete ${noun} "${item.name}"?`)) return
    setErr(null)
    try { await api(`${path}/${item.id}`, { method: 'DELETE' }); await load() } catch (x) { setErr(x) }
  }
  const fe = fieldErrors(err)

  return (
    <div className="space-y-5">
      <h1 className="text-2xl font-bold">{title}</h1>
      <form onSubmit={add} className="flex items-end gap-2">
        <div className="max-w-xs flex-1"><Field label={`New ${noun}`} value={name} onChange={(e) => setName(e.target.value)} error={fe.name} required /></div>
        <Btn disabled={busy}>Add {noun}</Btn>
      </form>
      <Alert>{err && !Object.keys(fe).length ? err.message : null}</Alert>
      <Card>
        {!items ? <Empty>Loading…</Empty> : items.length === 0 ? <Empty>No {noun}s yet. Add the first one above.</Empty> : (
          <ul className="divide-y divide-sand">
            {items.map((i) => (
              <li key={i.id} className="flex items-center justify-between px-4 py-3 text-sm">
                <span className="font-medium">{i.name}</span>
                <Btn variant="danger" onClick={() => remove(i)}>Delete</Btn>
              </li>
            ))}
          </ul>
        )}
      </Card>
    </div>
  )
}

  }

  const finalize = () => {
    if (!confirm(`Finalize ${month}? The month will be locked and nobody can change it.`)) return
    return act(() => api(`/payroll/runs/${month}/finalize`, { method: 'POST', body: {} }))
  }
  const pdf = (i) => download(`/payroll/payslip/${month}?employeeId=${i.employeeId}`, `payslip-${i.employeeCode}-${month}.pdf`).catch(setErr)

              <button key={x.month} onClick={() => setMonth(x.month)} className={`rounded border px-2 py-1 ${x.month === month ? 'border-pine bg-pine/10' : 'border-sand'}`}>
                {x.month} &middot; {x.status}
              </button>
            ))}
          </div>
        )}
        {!r && !run.loading && <p className="text-sm text-gray-500">{user.role === 'hr' ? 'No payroll for this month yet. Click Create draft.' : 'HR has not created a draft for this month yet.'}</p>}
        {r && (

            <p className={`mb-3 inline-block rounded px-2 py-0.5 text-xs font-medium ${locked ? 'bg-green-100 text-green-800' : 'bg-amber-100 text-amber-800'}`}>

import { useEffect, useState } from 'react'
import { api } from '../api.js'
import { useAuth } from '../auth.jsx'
import { Btn, Card, Err, Field, inputCls, useLoad } from '../components/bits.jsx'

const DAYS = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat']
const ZONES = (() => { try { return Intl.supportedValuesOf('timeZone') } catch { return ['UTC', 'Asia/Karachi'] } })()

export default function Settings() {
  const { user } = useAuth()
  const canEdit = user.role === 'admin'
  const { data, error, reload } = useLoad(() => api('/settings'), [])
  const [f, setF] = useState(null)
  const [err, setErr] = useState(null)
  const [ok, setOk] = useState(false)
  const [busy, setBusy] = useState(false)

  useEffect(() => {
    if (data) setF({ companyName: data.companyName, timezone: data.timezone, officeStart: data.officeStart, graceMinutes: data.graceMinutes, halfDayHours: data.halfDayHours, weekend: data.weekend, lat: data.office?.lat ?? '', lng: data.office?.lng ?? '', radiusM: data.office?.radiusM ?? 150 })
  }, [data])

  if (!f) return <div><Err error={error} /><p className="text-sm text-gray-400">Loading settings...</p></div>
  const set = (k, v) => { setOk(false); setF({ ...f, [k]: v }) }
  const toggleDay = (d) => set('weekend', f.weekend.includes(d) ? f.weekend.filter((x) => x !== d) : [...f.weekend, d])
  const here = () => navigator.geolocation?.getCurrentPosition((p) => setF({ ...f, lat: Number(p.coords.latitude.toFixed(6)), lng: Number(p.coords.longitude.toFixed(6)) }), () => setErr({ message: 'Could not read your location. Allow location access or type the numbers.' }), { enableHighAccuracy: true })

  const save = async (e) => {
    e.preventDefault()
    setBusy(true); setErr(null); setOk(false)
    const body = {
      companyName: f.companyName, timezone: f.timezone, officeStart: f.officeStart,
      graceMinutes: Number(f.graceMinutes), halfDayHours: Number(f.halfDayHours), weekend: f.weekend,
      office: { radiusM: Number(f.radiusM), ...(f.lat !== '' && f.lng !== '' ? { lat: Number(f.lat), lng: Number(f.lng) } : {}) },
    }
    try { await api('/settings', { method: 'PUT', body }); setOk(true); await reload() } catch (x) { setErr(x) } finally { setBusy(false) }
  }

  return (
    <div>
      <h1 className="mb-4 font-display text-2xl font-bold">Company settings</h1>
      <form onSubmit={save}>
        <fieldset disabled={!canEdit || busy}>
          <Card title="Company">
            <div className="grid gap-3 sm:grid-cols-2">
              <Field label="Company name"><input className={inputCls} value={f.companyName} onChange={(e) => set('companyName', e.target.value)} /></Field>
              <Field label="Timezone (decides late and absent)">
                <select className={inputCls} value={f.timezone} onChange={(e) => set('timezone', e.target.value)}>
                  {(ZONES.includes(f.timezone) ? ZONES : [f.timezone, ...ZONES]).map((z) => <option key={z}>{z}</option>)}
                </select>
              </Field>
            </div>
          </Card>
          <Card title="Office hours">
            <div className="grid gap-3 sm:grid-cols-3">
              <Field label="Office start"><input type="time" className={inputCls} value={f.officeStart} onChange={(e) => set('officeStart', e.target.value)} /></Field>
              <Field label="Grace (minutes)"><input type="number" min="0" max="120" className={inputCls} value={f.graceMinutes} onChange={(e) => set('graceMinutes', e.target.value)} /></Field>
              <Field label="Half day under (hours)"><input type="number" min="1" max="12" step="0.5" className={inputCls} value={f.halfDayHours} onChange={(e) => set('halfDayHours', e.target.value)} /></Field>
            </div>
            <div className="mt-3 text-sm">
              <span className="mb-1 block text-xs text-gray-600">Weekend</span>
              <div className="flex flex-wrap gap-3">{DAYS.map((d, i) => <label key={d} className="flex items-center gap-1"><input type="checkbox" checked={f.weekend.includes(i)} onChange={() => toggleDay(i)} />{d}</label>)}</div>
            </div>
          </Card>
          <Card title="Office location (for camera attendance)">
            <div className="grid gap-3 sm:grid-cols-3">
              <Field label="Latitude"><input type="number" step="any" className={inputCls} value={f.lat} onChange={(e) => set('lat', e.target.value)} /></Field>
              <Field label="Longitude"><input type="number" step="any" className={inputCls} value={f.lng} onChange={(e) => set('lng', e.target.value)} /></Field>
              <Field label="Radius (metres)"><input type="number" min="10" max="5000" className={inputCls} value={f.radiusM} onChange={(e) => set('radiusM', e.target.value)} /></Field>
            </div>
            {canEdit && <Btn type="button" variant="ghost" className="mt-3" onClick={here}>Use my current location</Btn>}
            <p className="mt-2 text-xs text-gray-500">Leave latitude and longitude empty to skip the location check.</p>
          </Card>
        </fieldset>
        <Err error={err} />
        {canEdit ? (
          <div className="flex items-center gap-3"><Btn disabled={busy}>Save settings</Btn>{ok && <span className="text-sm text-green-700">Saved. Changes apply to new records only.</span>}</div>
        ) : <p className="text-sm text-gray-500">Only Admin can change settings.</p>}
      </form>
    </div>
  )
}
