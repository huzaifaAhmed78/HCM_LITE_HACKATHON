import jwt from 'jsonwebtoken'
import { env } from '../config/env.js'
import { User } from '../models/User.js'
import { ApiError } from '../utils/ApiError.js'
import { asyncHandler } from '../utils/asyncHandler.js'

// Checks the token, then loads the user from the database on every request,
// so deactivating a login takes effect immediately.
// A user with a temporary password can only use routes marked allowPasswordChange.
export const authenticate = ({ allowPasswordChange = false } = {}) =>
  asyncHandler(async (req, res, next) => {
    const [type, token] = (req.headers.authorization || '').split(' ')
    if (type !== 'Bearer' || !token) throw new ApiError(401, 'Authentication required')

    let payload
    try {
      payload = jwt.verify(token, env.jwtSecret)
    } catch {
      throw new ApiError(401, 'Invalid or expired token')
    }

    const user = await User.findById(payload.sub)
    if (!user || !user.isActive) throw new ApiError(401, 'Account not found or deactivated')
    if (user.mustChangePassword && !allowPasswordChange) {
      throw new ApiError(403, 'You must change your temporary password first')
    }

    req.user = {
      id: String(user._id),
      role: user.role,
      employeeId: user.employeeId ? String(user.employeeId) : null,
      name: user.name,
      email: user.email,
    }
    next()
  })

import { ApiError } from '../utils/ApiError.js'

export function notFound(req, res, next) {
  next(new ApiError(404, `Route not found: ${req.method} ${req.originalUrl}`))
}

// Every error leaves the API in the same shape: { success: false, message, errors }
export function errorHandler(err, req, res, next) {
  let status = 500
  let message = 'Something went wrong on the server'
  let errors = []

  if (err instanceof ApiError) {
    status = err.status
    message = err.message
    errors = err.errors
  } else if (err.code === 11000) {
    status = 409
    const field = Object.keys(err.keyPattern || {})[0] || 'value'
    message = `Duplicate ${field}`
    errors = [{ field, message: `This ${field} already exists` }]
  } else if (err.name === 'CastError') {
    status = 400
    message = 'Invalid id format'
  } else {
    console.error(err) // unexpected: log it, but never leak details to the client
  }

  res.status(status).json({ success: false, message, errors })
}

// validate(zodSchema, 'body' | 'query' | 'params'). Replaces the input with the cleaned data.
export const validate = (schema, source = 'body') => (req, res, next) => {
  const result = schema.safeParse(req[source])
  if (!result.success) {
    const errors = result.error.issues.map((i) => ({ field: i.path.join('.'), message: i.message }))
    throw new ApiError(422, 'Validation failed', errors)
  }
  req[source] = result.data
  next()
}

import bcrypt from 'bcryptjs'
import jwt from 'jsonwebtoken'
import { env } from '../config/env.js'
import { User } from '../models/User.js'
import { ApiError } from '../utils/ApiError.js'
import { logAudit } from './auditService.js'

export const hashPassword = (plain) => bcrypt.hash(plain, 10)

export function signToken(user) {
  return jwt.sign(
    { role: user.role, employeeId: user.employeeId ? String(user.employeeId) : null },
    env.jwtSecret,
    { subject: String(user._id), expiresIn: env.jwtExpiresIn }
  )
}

export async function login({ email, password }) {
  const user = await User.findOne({ email })
  const valid = user && user.isActive && (await bcrypt.compare(password, user.passwordHash))
  if (!valid) throw new ApiError(401, 'Invalid email or password') // same message for every failure

  user.lastLoginAt = new Date()
  await user.save()
  return { token: signToken(user), user }
}

export async function changePassword(userId, { currentPassword, newPassword }) {
  const user = await User.findById(userId)
  if (!(await bcrypt.compare(currentPassword, user.passwordHash))) {
    throw new ApiError(422, 'Current password is incorrect', [{ field: 'currentPassword', message: 'Incorrect password' }])
  }
  if (currentPassword === newPassword) {
    throw new ApiError(422, 'New password must be different', [{ field: 'newPassword', message: 'Use a new password' }])
  }

  user.passwordHash = await hashPassword(newPassword)
  user.mustChangePassword = false
  await user.save()

  await logAudit({ actor: { id: user._id, role: user.role }, action: 'auth.change_password', entity: 'User', entityId: user._id })
  return { token: signToken(user), user }
}

import mongoose from 'mongoose'
import { ApiError } from '../utils/ApiError.js'
import { docView } from '../views/docView.js'
import { logAudit } from './auditService.js'

// Small shared CRUD for simple setup lists (departments, designations, leave types, holidays).
// Every create / update / delete is written to the audit log with before and after.
export function makeCrudService({ Model, entity, sort, checkRefs = async () => {} }) {
  const find = async (id) => {
    const doc = await Model.findById(id)
    if (!doc) throw new ApiError(404, `${entity} not found`)
    return doc
  }

  return {
    list: () => Model.find().sort(sort),

    async create(actor, data) {
      await checkRefs(data)
      const doc = await Model.create(data)
      await logAudit({ actor, action: `${entity}.create`, entity, entityId: doc._id, after: docView(doc) })
      return doc
    },

    async update(actor, id, data) {
      const doc = await find(id)
      await checkRefs(data)
      const before = docView(doc)
      doc.set(data)
      await doc.save()
      await logAudit({ actor, action: `${entity}.update`, entity, entityId: doc._id, before, after: docView(doc) })
      return doc
    },

      async remove(actor, id) {
      const doc = await find(id)
      const field = { Department: 'departmentId', Designation: 'designationId' }[Model.modelName]
      if (field && mongoose.models.Employee && (await mongoose.models.Employee.exists({ [field]: id })))
        throw new ApiError(409, `${entity} is in use by employees and cannot be deleted`)
      await doc.deleteOne()
      await logAudit({ actor, action: `${entity}.delete`, entity, entityId: id, before: docView(doc) })
    },
  }
}

const populate = [
  { path: 'departmentId', select: 'name' },
  { path: 'designationId', select: 'name' },

async function assertRefs({ departmentId, designationId }) {
  const errors = []
  if (departmentId && !(await mongoose.model('Department').exists({ _id: departmentId })))
    errors.push({ field: 'departmentId', message: 'Department not found' })
  if (designationId && !(await mongoose.model('Designation').exists({ _id: designationId })))
    errors.push({ field: 'designationId', message: 'Designation not found' })
  if (errors.length) throw new ApiError(422, 'Validation failed', errors)
}

async function nextCode() {
  const last = await Employee.findOne().sort({ createdAt: -1 }).select('employeeCode')
  const n = last ? parseInt(last.employeeCode.slice(4), 10) + 1 : 1
  return `EMP-${String(n).padStart(4, '0')}`
}

const findEmployee = async (id) => {
  const emp = await Employee.findById(id).populate(populate)
  if (!emp) throw new ApiError(404, 'Employee not found')
  return emp
}

// HR registers an employee. Login is created with a temporary password, returned ONCE.
export async function createEmployee(actor, input) {
  const email = input.email.toLowerCase()
  if (await User.exists({ email })) throw new ApiError(409, 'Duplicate email')
  await assertRefs(input)

  const temporaryPassword = generateTempPassword()
  const _id = new mongoose.Types.ObjectId()
  const user = await User.create({
    name: input.name,
    email,
    passwordHash: await hashPassword(temporaryPassword),
    role: 'employee',
    employeeId: _id,
    mustChangePassword: true,
  })

  let emp
  try {
    emp = await Employee.create({
      _id,
      userId: user._id,
      employeeCode: await nextCode(),
      name: input.name,
      email,
      phone: input.phone,
      address: input.address,
      departmentId: input.departmentId,
      designationId: input.designationId,
      joiningDate: input.joiningDate,
      basic: input.basic,
    })
  } catch (err) {
    await user.deleteOne() // do not leave a login without an employee
    throw err
  }

  emp = await emp.populate(populate)
  await logAudit({ actor, action: 'employee.create', entity: 'Employee', entityId: emp._id, after: employeeView(emp) })
  return { employee: employeeView(emp), temporaryPassword }
}

const escapeRe = (s) => s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')

export async function listEmployees({ q, departmentId, designationId, status, page, limit }) {
  const p = Math.max(1, Number(page) || 1)
  const l = Math.min(50, Math.max(1, Number(limit) || 10))
  const filter = {}
  if (departmentId) filter.departmentId = departmentId
  if (designationId) filter.designationId = designationId
  if (status) filter.status = status
  if (q) {
    const re = new RegExp(escapeRe(String(q)), 'i')
    filter.$or = [{ name: re }, { email: re }, { employeeCode: re }]
  }
  const [items, total] = await Promise.all([
    Employee.find(filter).populate(populate).sort({ createdAt: -1 }).skip((p - 1) * l).limit(l),
    Employee.countDocuments(filter),
  ])
  return { items: items.map(employeeView), total, page: p, pages: Math.ceil(total / l) }
}

export async function updateEmployee(actor, id, patch) {
  const emp = await findEmployee(id)
  await assertRefs(patch)
  const before = employeeView(emp)
  emp.set(patch)
  await emp.save()
  const userPatch = {}
  if (patch.status) userPatch.isActive = patch.status === 'active'
  if (patch.name) userPatch.name = patch.name
  if (Object.keys(userPatch).length) await User.findByIdAndUpdate(emp.userId, userPatch)
  const fresh = await findEmployee(id)
  await logAudit({ actor, action: 'employee.update', entity: 'Employee', entityId: emp._id, before, after: employeeView(fresh) })
  return employeeView(fresh)
}

import { Router } from 'express'
import { authenticate } from '../middleware/authenticate.js'
import { requireRole } from '../middleware/requireRole.js'
import { validate } from '../middleware/validate.js'
import { makeCrudController } from '../controllers/crudController.js'
import { departmentService, designationService, leaveTypeService, holidayService } from '../services/orgService.js'
import { idParams } from '../validators/common.js'
import * as v from '../validators/orgValidators.js'

// Admin and HR can read. Only Admin can change.
function crudRouter(service, label, createSchema, updateSchema) {
  const c = makeCrudController(service, label)
  const router = Router()
  router.use(authenticate())
  router.get('/', requireRole('admin', 'hr'), c.list)
  router.post('/', requireRole('admin'), validate(createSchema), c.create)
  router.put('/:id', requireRole('admin'), validate(idParams, 'params'), validate(updateSchema), c.update)
  router.delete('/:id', requireRole('admin'), validate(idParams, 'params'), c.remove)
  return router
}

export const leaveTypeRoutes = crudRouter(leaveTypeService, 'Leave type', v.leaveTypeCreate, v.leaveTypeUpdate)
